000448: Private Sub EvaluateContractResult( _
000449:     ByVal rowValues As Variant, _
000450:     ByVal sourceFile As String, _

000452:     ByRef resultValue As String, _

000454: 
000455:     Dim generatedAt As Date
000456:     Dim expiresAt As Date
000457:     Dim computedChecksum As String
